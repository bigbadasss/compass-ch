<%*
/*
  Compass：每日结束时使用的每日问题提示，来源于 Marshall Goldsmith 的《Triggers》。
  请在每日笔记上运行此模板，可使用 Templater 的插入模板命令或自定义快捷键。
  模板会依次询问每个问题，回答范围为 1 到 10；然后询问每个 habit_* 属性是否完成，
  并把答案写入笔记属性，不会在正文中插入内容。
  问题来自 系统/Compass 配置.md 的 questions 列表；只有列表缺失时才使用下方 FALLBACK。
  保持“我是否尽了最大努力”的提问框架，评价努力而不是结果。
*/
const FALLBACK = [
  ["dq_goals",         "今天，我是否尽力设定清晰的目标？"],
  ["dq_progress",      "今天，我是否尽力推动目标取得进展？"],
  ["dq_meaning",       "今天，我是否尽力寻找意义？"],
  ["dq_happy",         "今天，我是否尽力让自己快乐？"],
  ["dq_relationships", "今天，我是否尽力建立积极的人际关系？"],
  ["dq_engaged",       "今天，我是否尽力全身心投入？"],
];
const file = tp.config.target_file;
const cache = app.metadataCache.getFileCache(file) || {};
const fm = cache.frontmatter || {};
const cfg = app.metadataCache.getFileCache(app.vault.getAbstractFileByPath("系统/Compass 配置.md"))?.frontmatter || {};
const HB = cfg.habit_prefix || "habit_";
const QUESTIONS = Array.isArray(cfg.questions) && cfg.questions.length ? cfg.questions.map(q => typeof q === "string" ? [q, "今天，我是否尽力做到“" + q.replace(/^dq_/, "").replace(/[_-]+/g, " ") + "”？"] : [q.key, q.text]).filter(x => x[0] && x[1]) : FALLBACK;
const answers = {};
let cancelled = false;
for (const [key, q] of QUESTIONS) {
  const a = await tp.system.prompt(`${q}（1 = 完全没有尽力，10 = 已经尽最大努力）`, fm[key] ? String(fm[key]) : "");
  if (a === null) { cancelled = true; break; }
  const n = parseInt(a);
  if (!isNaN(n)) answers[key] = Math.min(10, Math.max(1, n));
}
if (!cancelled) {
  const habits = Object.keys(fm).filter(k => k.startsWith(HB));
  for (const h of habits) {
    const labels = { journal: "写日记", exercise: "锻炼", reading: "阅读" };
    const raw = h.slice(HB.length).replace(/[_-]+/g, " ");
    const nice = labels[raw] || raw;
    const pick = await tp.system.suggester(["是", "否"], [true, false], false, `今天完成了“${nice}”吗？`);
    if (pick === null) break;
    answers[h] = pick;
  }
}
if (Object.keys(answers).length) {
  await app.fileManager.processFrontMatter(file, f => { Object.assign(f, answers); });
  new Notice(`已将 ${Object.keys(answers).length} 项回答保存到 ${file.basename}`);
}
-%>
